MAFFT nucleotide alignments of baculovirus reference genes

Note

This report was generated using AI under general human direction. At the time of generation, the contents have not been comprehensively reviewed by a human analyst.

Introduction

This analysis creates independent MAFFT nucleotide alignments for the 38 baculovirus core genes and for the three-gene marker dataset consisting of polh/gran, lef-8, and lef-9. The individual alignments retain every sequence in their respective reference FASTA files. Genome-group eligibility and the selection of one sequence per genome group and gene are applied only when the alignments are concatenated.

The accession-to-genome-group assignments are read from the outputs of 01_accession_matching.qmd. That matching procedure is not repeated here. No distances, phylogenetic trees, or additional sequence searches are performed.

The core-gene nucleotide FASTA files are read from data/reference_core_genes/nt/. The separate polh/gran marker is read from data/reference_marker_genes/18_1_polhgran_alpha_beta_gamma.fna. Sequence selection

The detail tables from the accession-matching analysis are the sole source of accession-to-genome_group_id assignments. Only rows marked matched_unique are eligible for sequence selection. The complete core-gene and three-gene eligibility flags are read directly from dataset_eligibility.tsv.

The table multiple_reference_sequences.tsv identifies genome-group/gene combinations with more than one uniquely matched reference sequence. Selection is deterministic: a candidate whose accession is identified in the matching metadata as a GenBank accession is preferred to a RefSeq accession or another accession. If candidates remain at the same priority, their sequences must be identical; the lexicographically smallest accession is then selected. Different sequences at the same priority cause an error rather than an arbitrary choice.

parse_fasta <- function(path) {
  lines <- readLines(path, warn = FALSE)
  header_rows <- which(startsWith(lines, ">"))
  if (length(header_rows) == 0L || header_rows[1] != 1L) stop("Invalid FASTA file: ", path, call. = FALSE)
  records <- lapply(seq_along(header_rows), function(i) {
    start <- header_rows[i] + 1L
    end <- if (i < length(header_rows)) header_rows[i + 1L] - 1L else length(lines)
    header <- substring(lines[header_rows[i]], 2L)
    sequence <- paste(lines[start:end][nzchar(trimws(lines[start:end]))], collapse = "")
    if (!nzchar(sequence)) stop("Empty FASTA sequence in ", path, " header ", header, call. = FALSE)
    list(header = header, accession = sub("[[:space:]].*$", "", header), sequence = sequence)
  })
  headers <- vapply(records, `[[`, character(1), "header")
  if (anyDuplicated(headers)) stop("Duplicate FASTA headers in ", path, call. = FALSE)
  records
}

normalise_accession <- function(x) sub("\\.[0-9]+$", "", x)
metadata_accessions <- function(x) {
  x <- unlist(strsplit(paste(x, collapse = ";"), ";", fixed = TRUE), use.names = FALSE)
  unique(normalise_accession(x[nzchar(x) & !is.na(x)]))
}

candidate_priority <- function(row) {
  accession <- normalise_accession(row$Detected_Accession)
  if (accession %in% metadata_accessions(row$GenBank_Accession)) return(1L)
  if (accession %in% metadata_accessions(c(row$RefSeq_Accession, row$Matched_Metadata_Aliases))) return(2L)
  3L
}

select_sequences <- function(details, dataset, genes, eligible_ids) {
  details <- details[details$Match_Status == "matched_unique" & details$Core_Gene %in% genes & details$Genome_ID %in% eligible_ids, , drop = FALSE]
  selection_rows <- list()
  selected_rows <- list()
  if (nrow(details) == 0L) stop("No matched records available for dataset ", dataset, call. = FALSE)
  keys <- unique(paste(details$Genome_ID, details$Core_Gene, sep = "\r"))
  for (key in sort(keys)) {
    parts <- strsplit(key, "\r", fixed = TRUE)[[1]]
    group_id <- parts[1]; gene <- parts[2]
    rows <- details[details$Genome_ID == group_id & details$Core_Gene == gene, , drop = FALSE]
    priorities <- vapply(seq_len(nrow(rows)), function(i) candidate_priority(rows[i, , drop = FALSE]), integer(1))
    best <- which(priorities == min(priorities))
    records <- all_input_records[[rows$Reference_FASTA[best[1]]]]
    sequences <- vapply(best, function(i) {
      hit <- which(vapply(records, `[[`, character(1), "header") == rows$FASTA_Header[i])
      if (length(hit) != 1L) stop("Could not uniquely locate FASTA header in input: ", rows$FASTA_Header[i], call. = FALSE)
      records[[hit]]$sequence
    }, character(1))
    if (length(unique(sequences)) > 1L) {
      stop("Different sequences remain at the same selection priority for ", dataset, ", ", group_id, ", ", gene, call. = FALSE)
    }
    ordered <- best[order(normalise_accession(rows$Detected_Accession[best]))]
    chosen <- ordered[1]
    reason <- if (nrow(rows) == 1L) "single uniquely matched reference" else if (length(best) < nrow(rows)) "preferred higher-priority reference" else "identical same-priority references; accession tie-break"
    alternatives <- setdiff(rows$Detected_Accession, rows$Detected_Accession[chosen])
    selection_rows[[length(selection_rows) + 1L]] <- data.frame(
      dataset = dataset, gene = gene, genome_group_id = group_id,
      selected_accession = rows$Detected_Accession[chosen],
      alternative_accessions = paste(alternatives, collapse = ";"),
      selection_reason = reason, stringsAsFactors = FALSE
    )
    selected_rows[[length(selected_rows) + 1L]] <- rows[chosen, , drop = FALSE]
  }
  list(selection = do.call(rbind, selection_rows), selected = do.call(rbind, selected_rows))
}

Multiple sequence alignment

MAFFT is run independently for every core-gene input and again for each of the three marker inputs. The command uses nucleotide mode, automatic strategy selection, and four threads. Standard output is written directly to the alignment file; standard error is written to a temporary diagnostic file and is never treated as FASTA content. A non-zero return status stops the document.

mafft_path <- Sys.which("mafft")

run_mafft <- function(input, output) {
  error_file <- tempfile("mafft_stderr_")
  if (file.exists(output)) unlink(output)
  status <- system2(mafft_path, c("--nuc", "--auto", "--thread", "4", input), stdout = output, stderr = error_file)
  error_text <- if (file.exists(error_file)) paste(readLines(error_file, warn = FALSE), collapse = "\n") else ""
  unlink(error_file)
  if (!identical(status, 0L) || !file.exists(output) || file.info(output)$size == 0) {
    if (file.exists(output)) unlink(output)
    stop("MAFFT failed for ", input, " (status ", status, "): ", error_text, call. = FALSE)
  }
  invisible(output)
}

Alignment validation and statistics

Each alignment is checked against its original FASTA file. Header identity, sequence counts, non-empty sequences, sequence losses, and uniform alignment length are validated. Gap percentages are calculated over all cells of each alignment.

alignment_stats <- list()
alignment_records <- list()
validate_alignment <- function(input, alignment, dataset, gene) {
  input_records <- parse_fasta(input)
  aligned_records <- parse_fasta(alignment)
  input_headers <- vapply(input_records, `[[`, character(1), "header")
  aligned_headers <- vapply(aligned_records, `[[`, character(1), "header")
  lengths <- nchar(vapply(aligned_records, `[[`, character(1), "sequence"))
  if (length(input_records) != length(aligned_records) || !setequal(input_headers, aligned_headers)) stop("Alignment changed sequence count or FASTA headers: ", alignment, call. = FALSE)
  if (any(lengths == 0L) || length(unique(lengths)) != 1L) stop("Alignment sequences do not have one common non-empty length: ", alignment, call. = FALSE)
  input_lengths <- nchar(vapply(input_records, `[[`, character(1), "sequence"))
  sequence_matrix <- do.call(rbind, strsplit(vapply(aligned_records, `[[`, character(1), "sequence"), split = "", fixed = TRUE))
  alignment_records[[paste(dataset, gene, sep = "\r")]] <<- aligned_records
  data.frame(dataset = dataset, gene = gene, input_file = input, alignment_file = alignment,
             sequence_count = length(input_records), minimum_input_length = min(input_lengths),
             maximum_input_length = max(input_lengths), alignment_length = unique(lengths),
             gap_percentage = 100 * sum(sequence_matrix == "-") / length(sequence_matrix),
             mafft_parameters = mafft_parameters, stringsAsFactors = FALSE)
}

Concatenation

The core concatenation follows the numeric order of the input filenames. The three-gene concatenation follows polh/gran, lef-8, and lef-9. Sequence identity is joined exclusively through the selected genome_group_id. CuniNPV receives gap characters for the missing polh/gran partition only; no sequence is fabricated.

get_aligned_sequence <- function(dataset, gene, detail_row) {
  records <- alignment_records[[paste(dataset, gene, sep = "\r")]]
  hit <- which(vapply(records, `[[`, character(1), "header") == detail_row$FASTA_Header)
  if (length(hit) != 1L) stop("Selected header not found in alignment for ", dataset, ": ", detail_row$FASTA_Header, call. = FALSE)
  records[[hit]]$sequence
}

build_concatenation <- function(dataset, genes, eligible_ids, selected_details, output, partition_file) {
  selected_details <- selected_details[selected_details$Genome_ID %in% eligible_ids, , drop = FALSE]
  lengths <- vapply(genes, function(g) alignment_statistics$alignment_length[alignment_statistics$dataset == dataset & alignment_statistics$gene == g], integer(1))
  partitions <- data.frame(gene = genes, source_alignment = character(length(genes)), start = integer(length(genes)), end = integer(length(genes)), alignment_length = lengths, stringsAsFactors = FALSE)
  partitions$source_alignment <- vapply(genes, function(g) file.path(if (dataset == "core_38") core_individual_dir else three_individual_dir, if (dataset == "core_38") core_files[match(g, core_gene_names)] else three_output_names[g]), character(1))
  partitions$start <- cumsum(c(1L, head(lengths, -1L)))
  partitions$end <- cumsum(lengths)
  groups <- sort(eligible_ids)
  output_records <- character()
  for (group_id in groups) {
    pieces <- character(length(genes))
    for (i in seq_along(genes)) {
      gene <- genes[i]
      row <- selected_details[selected_details$Genome_ID == group_id & selected_details$Core_Gene == gene, , drop = FALSE]
      if (nrow(row) == 0L) {
        if (dataset == "three_genes" && group_id == cuni_group && gene == "polh_gran") pieces[i] <- paste(rep("-", lengths[i]), collapse = "") else stop("Missing selected sequence for ", dataset, ", ", group_id, ", ", gene, call. = FALSE)
      } else if (nrow(row) != 1L) stop("More than one selected sequence for ", dataset, ", ", group_id, ", ", gene, call. = FALSE) else pieces[i] <- get_aligned_sequence(dataset, gene, row)
    }
    output_records <- c(output_records, paste0(">", group_id), pieces[[1]])
    if (length(pieces) > 1L) output_records[length(output_records)] <- paste0(pieces, collapse = "")
  }
  if (length(groups) == 0L) stop("No eligible genome groups for ", dataset, call. = FALSE)
  writeLines(output_records, output, useBytes = TRUE)
  write.table(partitions, partition_file, sep = "\t", quote = FALSE, row.names = FALSE, na = "")
  list(partitions = partitions, groups = groups, length = sum(lengths))
}

Validation

The final checks validate FASTA format, uniqueness of concatenated genome-group headers, common concatenation lengths, exact partition coverage, complete core-gene coverage, three-gene eligibility, and the CuniNPV gap-only polh/gran partition.

validate_concat <- function(path, expected_groups, partitions) {
  records <- parse_fasta(path)
  headers <- vapply(records, `[[`, character(1), "header")
  lengths <- nchar(vapply(records, `[[`, character(1), "sequence"))
  if (!setequal(headers, expected_groups) || anyDuplicated(headers)) stop("Concatenated FASTA headers are invalid: ", path, call. = FALSE)
  if (length(unique(lengths)) != 1L || unique(lengths) != sum(partitions$alignment_length)) stop("Concatenation length validation failed: ", path, call. = FALSE)
  if (!identical(partitions$start, c(1L, head(partitions$end, -1L) + 1L)) || partitions$end[nrow(partitions)] != unique(lengths)) stop("Partition validation failed: ", path, call. = FALSE)
  invisible(records)
}

three_records <- validate_concat(file.path(three_concat_dir, "three_genes.fna"), three_ids, three_concat$partitions)
