Transition and transversion patterns: 38 Baculovirus core genes

This is an independent step-04 analysis of the existing 38-core-gene concatenated alignment. No alignment, sequence selection, or MAFFT analysis is performed. The requested results/alignments/core_38/ alias is checked first; the current step-02 document actually writes the same alignment to results/alignments/core_genes/, which is used as an explicit fallback.

read_fasta <- function(path) {
  lines <- readLines(path, warn = FALSE); h <- which(startsWith(lines, ">"))
  if (!length(h) || h[1] != 1L) stop("Invalid FASTA: ", path, call. = FALSE)
  out <- lapply(seq_along(h), function(i) { a <- h[i] + 1L; b <- if (i < length(h)) h[i + 1L] - 1L else length(lines); s <- paste(lines[a:b][nzchar(trimws(lines[a:b]))], collapse = ""); if (!nzchar(s)) stop("Empty FASTA sequence.", call. = FALSE); list(header = substring(lines[h[i]], 2L), sequence = s) })
  ids <- vapply(out, `[[`, character(1), "header"); if (anyDuplicated(ids)) stop("Duplicate FASTA headers.", call. = FALSE); out
}

Pairwise substitution calculations

Each pair is evaluated directly from the alignment. Only paired A/C/G/T positions are retained. The K2P expression is the same as in the original step 04 and returns NA when either logarithm is undefined.

pairwise_substitutions <- function(sequences, genome_ids) {
  raw <- matrix(unlist(lapply(toupper(sequences), charToRaw), use.names = FALSE), nrow = length(sequences), byrow = TRUE); codes <- as.raw(charToRaw("ACGT")); ind <- lapply(codes, function(x) raw == x); valid <- Reduce(`|`, ind)
  compared_matrix <- tcrossprod(valid * 1L); identical_matrix <- Reduce(`+`, lapply(ind, function(x) tcrossprod(x * 1L)))
  transitions_matrix <- tcrossprod(ind[[1]] * 1L, ind[[3]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[2]] * 1L)
  transversions_matrix <- tcrossprod(ind[[1]] * 1L, ind[[2]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[1]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[2]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[3]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[3]] * 1L)
  ij <- which(upper.tri(compared_matrix), arr.ind = TRUE); compared <- as.integer(compared_matrix[ij]); identical <- as.integer(identical_matrix[ij]); transitions <- as.integer(transitions_matrix[ij]); transversions <- as.integer(transversions_matrix[ij]); P <- ifelse(compared > 0, transitions / compared, NA_real_); Q <- ifelse(compared > 0, transversions / compared, NA_real_); k2p <- ifelse(compared > 0 & 1 - 2 * P - Q > 0 & 1 - 2 * Q > 0, -.5 * log(1 - 2 * P - Q) - .25 * log(1 - 2 * Q), NA_real_)
  data.frame(genome_group_id_1 = genome_ids[ij[,1]], genome_group_id_2 = genome_ids[ij[,2]], compared_positions = compared, identical_positions = identical, transitions = transitions, transversions = transversions, transition_frequency = P, transversion_frequency = Q, observed_divergence = P + Q, identity_percentage = ifelse(compared > 0, 100 * identical / compared, NA_real_), ts_tv_ratio = ifelse(transversions > 0, transitions / transversions, NA_real_), k2p_distance = k2p, stringsAsFactors = FALSE)
}
substitution_patterns <- pairwise_substitutions(sequences, genome_ids)
new_keys <- paste(substitution_patterns$genome_group_id_1, substitution_patterns$genome_group_id_2, sep = "\r")
if (!identical(new_keys, identity_keys)) stop("Pair order differs from step 03.", call. = FALSE)
if (any(substitution_patterns$identical_positions + substitution_patterns$transitions + substitution_patterns$transversions != substitution_patterns$compared_positions)) stop("Counts do not sum to compared positions.", call. = FALSE)
if (any(abs(substitution_patterns$identity_percentage - identity_table$identity_percentage) > 1e-10, na.rm = TRUE)) stop("Identity values disagree with step 03.", call. = FALSE)
if (any(abs(substitution_patterns$observed_divergence - (1 - substitution_patterns$identity_percentage / 100)) > 1e-12, na.rm = TRUE)) stop("Observed divergence is inconsistent with identity.", call. = FALSE)
if (any(substitution_patterns$transition_frequency < 0 | substitution_patterns$transition_frequency > 1 | substitution_patterns$transversion_frequency < 0 | substitution_patterns$transversion_frequency > 1, na.rm = TRUE)) stop("Invalid frequencies.", call. = FALSE)
write.table(substitution_patterns, file.path(output_dir, "pairwise_substitution_patterns.tsv"), sep = "\t", quote = FALSE, row.names = FALSE, na = "")
