read_fasta <- function(path) {
  lines <- readLines(path, warn = FALSE)
  h <- which(startsWith(lines, ">"))
  if (!length(h) || h[1] != 1L) stop("Invalid FASTA: ", path, call. = FALSE)
  out <- lapply(seq_along(h), function(i) {
    a <- h[i] + 1L; b <- if (i < length(h)) h[i + 1L] - 1L else length(lines)
    s <- paste(lines[a:b][nzchar(trimws(lines[a:b]))], collapse = "")
    if (!nzchar(s)) stop("Empty FASTA sequence in ", path, call. = FALSE)
    list(header = substring(lines[h[i]], 2L), sequence = s)
  })
  ids <- vapply(out, `[[`, character(1), "header")
  if (anyDuplicated(ids)) stop("Duplicate FASTA headers in ", path, call. = FALSE)
  out
}

Pairwise substitution calculations

Only paired A, C, G, and T positions are used. Gaps and ambiguous symbols are excluded. K2P is calculated from the observed nucleotide counts, not from the identity table; it is NA whenever a logarithm is not mathematically defined.

pairwise_substitutions <- function(sequences, genome_ids) {
  raw <- matrix(unlist(lapply(toupper(sequences), charToRaw), use.names = FALSE), nrow = length(sequences), byrow = TRUE)
  codes <- as.raw(charToRaw("ACGT"))
  ind <- lapply(codes, function(x) raw == x)
  valid <- Reduce(`|`, ind)
  compared_matrix <- tcrossprod(valid * 1L)
  identical_matrix <- Reduce(`+`, lapply(ind, function(x) tcrossprod(x * 1L)))
  transition_matrix <- tcrossprod(ind[[1]] * 1L, ind[[3]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[2]] * 1L)
  transversion_matrix <- tcrossprod(ind[[1]] * 1L, ind[[2]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[1]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[1]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[2]] * 1L) + tcrossprod(ind[[2]] * 1L, ind[[3]] * 1L) + tcrossprod(ind[[3]] * 1L, ind[[4]] * 1L) + tcrossprod(ind[[4]] * 1L, ind[[3]] * 1L)
  ij <- which(upper.tri(compared_matrix), arr.ind = TRUE)
  compared <- as.integer(compared_matrix[ij]); identical <- as.integer(identical_matrix[ij]); transitions <- as.integer(transition_matrix[ij]); transversions <- as.integer(transversion_matrix[ij])
  P <- ifelse(compared > 0, transitions / compared, NA_real_); Q <- ifelse(compared > 0, transversions / compared, NA_real_)
  k2p <- ifelse(compared > 0 & (1 - 2 * P - Q) > 0 & (1 - 2 * Q) > 0, -0.5 * log(1 - 2 * P - Q) - 0.25 * log(1 - 2 * Q), NA_real_)
  data.frame(genome_group_id_1 = genome_ids[ij[, 1]], genome_group_id_2 = genome_ids[ij[, 2]], compared_positions = compared, identical_positions = identical, transitions = transitions, transversions = transversions, transition_frequency = P, transversion_frequency = Q, observed_divergence = P + Q, identity_percentage = ifelse(compared > 0, 100 * identical / compared, NA_real_), ts_tv_ratio = ifelse(transversions > 0, transitions / transversions, NA_real_), k2p_distance = k2p, stringsAsFactors = FALSE)
}
